227def main():
228    parser = buildArgsParser()
229    args = parser.parse_args()
230
231    block_size = np.array(args.spatial_block_size + (args.angular_block_size,))
232    logger = logging.getLogger('nlsam')
233    noise_method = args.noise_method
234    verbose = args.verbose
235
236    if args.logfile is not None:
237        handler = logging.FileHandler(args.logfile)
238        verbose = True
239    else:
240        handler = logging.StreamHandler(args.logfile)
241
242    formatter = logging.Formatter('%(asctime)s - %(name)s - %(levelname)s - %(message)s', "%Y-%m-%d %H:%M")
243    handler.setFormatter(formatter)
244    logger.addHandler(handler)
245
246    if verbose:
247        logger.setLevel(logging.INFO)
248        logger.info('Verbosity is on')
249
250    if args.no_stabilization:
251        logger.info('Stabilization disabled!')
252
253    if args.no_denoising:
254        logger.info('Denoising disabled!')
255
256    if args.use_f32:
257        logger.info('Computations will be made using float32!')
258        dtype = np.float32
259    else:
260        dtype = np.float64
261
262    ##########################################
263    #  Load up data and do some sanity checks
264    ##########################################
265
266    overwritable_files = [args.output,
267                          args.save_sigma,
268                          args.save_difference,
269                          args.save_piesno_mask,
270                          args.save_stabilization,
271                          args.save_eta]
272
273    for f in overwritable_files:
274        if f is not None and os.path.isfile(f):
275            if args.overwrite:
276                logger.warning(f'Overwriting {os.path.realpath(f)}')
277            else:
278                parser.error(f'{f} already exists! Use -f or --force to overwrite it.')
279
280    # Load up N or turn it into a number if it fails
281    if args.N == 'auto':
282        N = 'auto'
283    else:
284        try:
285            N = nib.load(args.N).get_fdata(caching='unchanged', dtype=np.float32)
286        except IOError:
287            N = float(args.N)
288
289    # this is to prevent triggering an error check since noise_method is not used in this case and skipped in an else-if clause
290    if args.load_sigma is not None:
291        noise_method = None
292
293    if args.N != 'auto':
294        if args.noise_maps is not None:
295            parser.error(f'You need to pass -N auto when using noise maps, but you passed -N {args.N}')
296
297        if noise_method == 'auto' or noise_method == 'auto_maxlk':
298            parser.error(f'You need to pass -N auto when using --noise_est auto/auto_maxlk, but you passed -N {args.N}')
299
300    if args.noise_maps is None:
301        if args.N == 'auto' and noise_method not in ['auto', 'auto_maxlk']:
302            parser.error(f'You need to pass --noise_est auto/auto_maxlk when using -N auto, but you passed --noise_est {noise_method}. Pass -N explicitly or use --noise_est {args.N}')
303
304    if args.load_sigma is None and isinstance(N, np.ndarray):
305        parser.error(f'You need to pass --load_sigma sigma.nii.gz when loading N as a volume, but you passed -N {args.N}')
306
307    if args.load_sigma is not None and args.N == 'auto':
308        parser.error(f'You need to pass -N explicitly when using --load_sigma sigma.nii.gz, but you passed -N {args.N}')
309
310    vol = nib.load(args.input)
311    data = vol.get_fdata(caching='unchanged', dtype=np.float32)
312    affine = vol.affine
313    header = vol.header
314    header.set_data_dtype(np.float32)
315    logger.info(f"Loading data {os.path.realpath(args.input)}")
316
317    if args.mask is not None:
318        mask = nib.load(args.mask).get_fdata(caching='unchanged').astype(bool)
319        logger.info(f"Loading mask {os.path.realpath(args.mask)}")
320    else:
321        mask = np.ones(data.shape[:-1], dtype=bool)
322
323    # load m_hat if we supplied the argument
324    if args.load_mhat is not None:
325        m_hat = nib.load(args.load_mhat).get_fdata(caching='unchanged', dtype=np.float32)
326        logger.info(f"Loading m_hat {os.path.realpath(args.load_mhat)}")
327        if m_hat.shape != data.shape:
328            raise ValueError(f'm_hat shape {m_hat.shape} is different from data shape {data.shape}!')
329    else:
330        m_hat = data
331
332    bvals, bvecs = read_bvals_bvecs(args.bvals, args.bvecs)
333
334    if args.cores is None or args.cores > os.cpu_count():
335        n_cores = os.cpu_count()
336    else:
337        n_cores = args.cores
338
339    if len(block_size) != len(data.shape):
340        raise ValueError(f'Block shape {block_size} and data shape {data.shape} are not of the same length')
341
342    if data.shape[:-1] != mask.shape:
343        raise ValueError(f'data shape is {data.shape}, but mask shape {mask.shape} is different!')
344
345    if args.gamma <= 0:
346        raise ValueError(f'gamma should be positive but has value {args.gamma}')
347
348    #########################
349    #  Noise estimation part
350    #########################
351
352    if args.load_sigma is not None:
353        sigma = nib.load(args.load_sigma).get_fdata(caching='unchanged', dtype=np.float32)
354        logger.info(f"Loading sigma {os.path.realpath(args.load_sigma)}")
355
356        # If we have a bunch of 4D views, we only want a 3D array
357        if sigma.ndim == data.ndim:
358            sigma = np.median(sigma, axis=-1)
359            logger.warning(f"{os.path.realpath(args.load_sigma)} was {sigma.ndim + 1}D, but was downcasted to {sigma.ndim}D.")
360
361        if data.shape[:-1] != sigma.shape:
362            raise ValueError(f'data shape is {data.shape}, but sigma shape {sigma.shape} is different!')
363
364    elif args.noise_maps is not None:
365        noise_maps = nib.load(args.noise_maps).get_fdata(caching='unchanged', dtype=np.float32)
366        logger.info(f"Loading {os.path.realpath(args.noise_maps)} as a noise map")
367
368        # Needs to be 4D
369        if noise_maps.ndim != 4:
370            noise_maps = noise_maps[..., None]
371
372        sigma, N, mask_noise = estimate_from_nmaps(noise_maps, ncores=n_cores, verbose=verbose)
373
374        if args.save_piesno_mask is not None:
375            nib.save(nib.Nifti1Image(mask_noise.astype(np.int16), affine), args.save_piesno_mask)
376            logger.info(f"Mask of noisy voxels saved as {os.path.realpath(args.save_piesno_mask)}")
377
378        if args.save_N is not None:
379            nib.save(nib.Nifti1Image(N, affine), args.save_N)
380            logger.info(f"N map saved as {os.path.realpath(args.save_N)}")
381
382    elif noise_method == 'local_std':
383        logger.info(f"Estimating noise with method {noise_method}")
384        sigma = local_standard_deviation(data, n_cores=n_cores, verbose=verbose)
385
386        # Compute the corrected value for each 3D volume
387        if N > 0:
388            sigma = root_finder_sigma(data, sigma, N, mask=mask, verbose=verbose, n_cores=n_cores)
389
390    elif noise_method == 'auto' or noise_method == 'auto_maxlk':
391        logger.info(f"Estimating noise with method {noise_method}")
392
393        method = 'moments' if noise_method == 'auto' else 'maxlk'
394        sigma_1D, N_1D, mask_noise = estimate_from_dwis(data, return_mask=True, ncores=n_cores, verbose=verbose, method=method)
395        sigma = np.broadcast_to(sigma_1D[None, None, :, None], data.shape)
396        N = np.broadcast_to(N_1D[None, None, :, None], data.shape)
397
398        if args.save_piesno_mask is not None:
399            nib.save(nib.Nifti1Image(mask_noise.astype(np.int16), affine), args.save_piesno_mask)
400            logger.info(f"Mask of noisy voxels saved as {os.path.realpath(args.save_piesno_mask)}")
401
402        if args.save_N is not None:
403            nib.save(nib.Nifti1Image(N, affine), args.save_N)
404            logger.info(f"N map saved as {os.path.realpath(args.save_N)}")
405
406    if args.save_sigma is not None:
407        nib.save(nib.Nifti1Image(sigma, affine), args.save_sigma)
408        logger.info(f"Sigma map saved as {os.path.realpath(args.save_sigma)}")
409
410    ##################
411    #  Stabilizer part
412    ##################
413
414    if args.no_stabilization:
415        data_stabilized = data
416    else:
417        logger.info("Now performing stabilization")
418
419        # We may have a 3D sigma map, so broadcast to 4D for indexing
420        if sigma.ndim == 3:
421            sigma = np.broadcast_to(sigma[..., None], data.shape)
422
423        data_stabilized, eta = stabilization(data,
424                                             m_hat,
425                                             sigma,
426                                             N,
427                                             mask=mask,
428                                             clip_eta=args.no_clip_eta,
429                                             return_eta=True,
430                                             n_cores=n_cores,
431                                             verbose=verbose)
432
433        if args.save_eta is not None:
434            nib.save(nib.Nifti1Image(eta, affine), args.save_eta)
435            logger.info(f"eta volume saved as {os.path.realpath(args.save_eta)}")
436
437        if args.save_stabilization is not None:
438            nib.save(nib.Nifti1Image(data_stabilized, affine), args.save_stabilization)
439            logger.info(f"Stabilized data saved as {os.path.realpath(args.save_stabilization)}")
440
441        del m_hat, eta
442
443    ##################
444    #  Denoising part
445    ##################
446
447    if not args.no_denoising:
448        logger.info(f"Now denoising {os.path.realpath(args.input)}")
449
450        # If we have a bunch of 4D views, we only want a 3D array
451        if sigma.ndim == data.ndim:
452            sigma = np.median(sigma, axis=-1)
453
454        if args.save_difference is None:
455            del data
456
457        all_args = {
458            'mask': mask,
459            'is_symmetric': args.is_symmetric,
460            'n_cores': n_cores,
461            'split_b0s': args.split_b0s,
462            'split_shell': args.split_shell,
463            'subsample': args.no_subsample,
464            'n_iter': args.iterations,
465            'b0_threshold': args.b0_threshold,
466            'bval_threshold': args.bval_threshold,
467            'dtype': dtype,
468            'verbose': verbose,
469            'gamma': args.gamma
470        }
471
472        data_denoised = nlsam_denoise(data_stabilized,
473                                      sigma,
474                                      bvals,
475                                      bvecs,
476                                      block_size,
477                                      **all_args)
478
479        nib.save(nib.Nifti1Image(data_denoised.astype(np.float32), affine, header), args.output)
480        logger.info(f"Denoised data saved as {os.path.realpath(args.output)}")
481
482        if args.save_difference is not None:
483            nib.save(nib.Nifti1Image(np.abs(data_denoised - data).astype(np.float32), affine, header), args.save_difference)
484            logger.info(f"Difference map saved as {os.path.realpath(args.save_difference)}")
485
486    return None

489def main_workaround_joblib():
490    # Until joblib.loky support pyinstaller, we use threading instead for the frozen binaries
491    import sys
492    frozen = getattr(sys, 'frozen', False) and hasattr(sys, '_MEIPASS')
493
494    if frozen:
495        from joblib import parallel_config
496        with parallel_config(backend="threading"):
497            main()
498    else:
499        main()
500
501# This is needed for pyinstaller since it requires a file, even if we have an entrypoint in the wheel version
502if __name__ == "__main__":
503    main_workaround_joblib()
